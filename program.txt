# Use of variables is work in progress
.data
variable: .word 12
blank_space: .space 40 # Oh great, I'm going to get called a Swiftie for naming a variable "blank space"
.text

# Memory[12] holds the value 123

# Load Memory[12]
addi $s1, $s1, 12
lw $s0, 0($s1)

# Why not, just for fun
sll $s2, $s0, 1

# store $s2 back into memory
addi $t0, $t0, 24
sw $s2, 0($t0)
sw $t0, 0($zero)

# Do 246 * 10 and store it
li $s3, 10
mult $s3, $s2
mfhi $t2
mflo $t3
add $t1, $t2, $t3
sw $t1, 64($zero)
